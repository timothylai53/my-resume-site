/**
 * SaveSnapshot - saves this month's Medium values into tblSnapshots.
 * 1. Takes the rows of med_Long where Source = "This month (not yet saved)"
 * 2. Removes any rows of the same month already in tblSnapshots (so a month can be redone)
 * 3. Adds this month's rows to tblSnapshots
 * How to use: Automate > New Script > select all > paste this > Save script > Run
 */
function main(workbook: ExcelScript.Workbook) {
  const src = workbook.getTable("med_Long");
  const dst = workbook.getTable("tblSnapshots");
  if (!src || !dst) {
    console.log("Table med_Long or tblSnapshots not found. Refresh the queries first.");
    return;
  }

  const srcHeaders = src.getHeaderRowRange().getValues()[0].map(v => String(v));
  const dstHeaders = dst.getHeaderRowRange().getValues()[0].map(v => String(v));
  const map = dstHeaders.map(h => srcHeaders.indexOf(h));
  const iSource = srcHeaders.indexOf("Source");
  if (iSource < 0 || map.some(i => i < 0)) {
    console.log("The column headers of med_Long and tblSnapshots do not match. Check the tblSnapshots headers.");
    return;
  }

  const srcBody = src.getRangeBetweenHeaderAndTotal().getValues();
  const newRows = srcBody
    .filter(r => String(r[iSource]).indexOf("This month") === 0)
    .map(r => map.map(i => r[i]));
  if (newRows.length === 0) {
    console.log("No rows for this month in med_Long. Refresh vm_Processed, then med_Long, then Refresh All.");
    return;
  }

  const dMonth = dstHeaders.indexOf("SnapshotMonth");
  const dKey = dstHeaders.indexOf("Key");
  const month = monthKey(newRows[0][dMonth]);

  // Keep other months and non-blank rows
  const oldRows = dst.getRangeBetweenHeaderAndTotal().getValues();
  const keep = oldRows.filter(r => monthKey(r[dMonth]) !== month && String(r[dKey]).trim() !== "");
  const replaced = oldRows.length - keep.length;

  const allRows = keep.concat(newRows).map(r => r.map((v, i) => (i === dMonth ? monthKey(v) : v)));

  // Resize the table (at least one row always stays)
  const count = dst.getRowCount();
  if (count > allRows.length) {
    dst.deleteRowsAt(allRows.length, count - allRows.length);
  } else if (count < allRows.length) {
    dst.addRows(-1, allRows.slice(count));
  }

  // Text format first, so Excel does not turn "2026-10" or "Feb-2027" into dates
  for (const name of ["SnapshotMonth", "Key", "Planned Remediation (Month)"]) {
    dst.getColumnByName(name).getRangeBetweenHeaderAndTotal().setNumberFormat("@");
  }
  dst.getRangeBetweenHeaderAndTotal().setValues(allRows);

  console.log(`Saved ${month}: ${newRows.length} rows written. tblSnapshots now has ${allRows.length} rows (${replaced} old rows of the same month or blank rows removed).`);
}

function monthKey(v: string | number | boolean): string {
  if (typeof v === "number") {
    // Excel date serial -> yyyy-MM
    const d = new Date(Math.round((v - 25569) * 86400 * 1000));
    return d.getUTCFullYear() + "-" + String(d.getUTCMonth() + 1).padStart(2, "0");
  }
  return String(v).replace(/^'/, "").trim();
}
